Lập trình bất đồng bộ trong JavaScript: Promise và Async/Await

October 14, 2025 · 3 min · Phú Sơn

Lập trình bất đồng bộ là một trong những khái niệm quan trọng nhất trong JavaScript. Hãy cùng tìm hiểu cách xử lý async hiệu quả.

Callback Hell - Vấn đề cần giải quyết#

// Code khó đọc và maintain
getUserData(userId, function(user) {
    getOrders(user.id, function(orders) {
        getOrderDetails(orders[0].id, function(details) {
            console.log(details);
        });
    });
});

Promise - Giải pháp đầu tiên#

// Tạo Promise
function fetchUser(userId) {
    return new Promise((resolve, reject) => {
        setTimeout(() => {
            if (userId > 0) {
                resolve({ id: userId, name: "Nguyễn Văn A" });
            } else {
                reject("User ID không hợp lệ");
            }
        }, 1000);
    });
}

// Sử dụng Promise
fetchUser(1)
    .then(user => {
        console.log("User:", user);
        return fetchOrders(user.id);
    })
    .then(orders => {
        console.log("Orders:", orders);
    })
    .catch(error => {
        console.error("Lỗi:", error);
    })
    .finally(() => {
        console.log("Hoàn thành");
    });

Async/Await - Cách viết hiện đại#

async function getUserOrders(userId) {
    try {
        const user = await fetchUser(userId);
        console.log("User:", user);
        
        const orders = await fetchOrders(user.id);
        console.log("Orders:", orders);
        
        const details = await fetchOrderDetails(orders[0].id);
        console.log("Details:", details);
        
        return details;
    } catch (error) {
        console.error("Lỗi:", error);
    } finally {
        console.log("Hoàn thành");
    }
}

// Gọi hàm async
getUserOrders(1);

Xử lý nhiều Promise cùng lúc#

Promise.all - Chờ tất cả hoàn thành#

async function fetchAllData() {
    const [users, products, orders] = await Promise.all([
        fetchUsers(),
        fetchProducts(),
        fetchOrders()
    ]);
    
    console.log("Tất cả dữ liệu:", { users, products, orders });
}

Promise.race - Lấy kết quả nhanh nhất#

async function getFastestServer() {
    const result = await Promise.race([
        fetch('https://server1.com/api/data'),
        fetch('https://server2.com/api/data'),
        fetch('https://server3.com/api/data')
    ]);
    
    return result;
}

Promise.allSettled - Chờ tất cả (kể cả lỗi)#

async function fetchWithErrors() {
    const results = await Promise.allSettled([
        fetchUsers(),
        fetchProducts(),
        Promise.reject("Lỗi server")
    ]);
    
    results.forEach((result, index) => {
        if (result.status === 'fulfilled') {
            console.log(`Request ${index} thành công:`, result.value);
        } else {
            console.log(`Request ${index} thất bại:`, result.reason);
        }
    });
}

Ví dụ thực tế: Fetch API#

async function getWeather(city) {
    try {
        const response = await fetch(`https://api.weather.com/${city}`);
        
        if (!response.ok) {
            throw new Error(`HTTP error! status: ${response.status}`);
        }
        
        const data = await response.json();
        return data;
    } catch (error) {
        console.error("Không thể lấy dữ liệu thời tiết:", error);
    }
}

// Sử dụng
getWeather("Hanoi").then(weather => {
    console.log("Thời tiết Hà Nội:", weather);
});

Error Handling tốt nhất#

async function robustFetch(url) {
    try {
        const response = await fetch(url);
        const data = await response.json();
        return { success: true, data };
    } catch (error) {
        return { success: false, error: error.message };
    }
}

// Sử dụng
const result = await robustFetch('https://api.example.com/data');
if (result.success) {
    console.log(result.data);
} else {
    console.error(result.error);
}

Kết luận#

Async/Await làm code bất đồng bộ dễ đọc và dễ maintain hơn rất nhiều. Hãy luôn sử dụng try-catch để xử lý lỗi!
