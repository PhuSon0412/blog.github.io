JavaScript cơ bản: Từ biến đến hàm

January 16, 2024 · 3 min · Phú Sơn

JavaScript là ngôn ngữ lập trình phổ biến nhất cho web development. Hãy cùng tìm hiểu các khái niệm cơ bản.

Khai báo biến: var, let, const#

// var - scope function, có hoisting
var x = 10;

// let - scope block, không hoisting
let y = 20;

// const - hằng số, không thay đổi được
const PI = 3.14159;

Nên dùng gì? Ưu tiên const, sau đó let, tránh var.

Các kiểu dữ liệu#

// Primitive types
let number = 42;              // Number
let text = "Xin chào";        // String
let isTrue = true;            // Boolean
let nothing = null;           // Null
let notDefined;               // Undefined
let symbol = Symbol("id");    // Symbol
let bigNum = 9007199254740991n; // BigInt

// Reference type
let person = {                // Object
    name: "An",
    age: 25
};
let colors = ["đỏ", "xanh", "vàng"]; // Array

Hàm (Functions)#

Function Declaration#

function greet(name) {
    return `Xin chào, ${name}!`;
}

console.log(greet("Minh")); // "Xin chào, Minh!"

Function Expression#

const add = function(a, b) {
    return a + b;
};

console.log(add(5, 3)); // 8

Arrow Function (ES6)#

const multiply = (a, b) => a * b;

const square = x => x * x;

const sayHello = () => console.log("Hello!");

Câu lệnh điều kiện#

let score = 85;

if (score >= 90) {
    console.log("Xuất sắc");
} else if (score >= 80) {
    console.log("Giỏi");
} else if (score >= 70) {
    console.log("Khá");
} else {
    console.log("Trung bình");
}

// Ternary operator
let result = score >= 50 ? "Đậu" : "Rớt";

Vòng lặp#

// For loop
for (let i = 0; i < 5; i++) {
    console.log(i);
}

// While loop
let count = 0;
while (count < 3) {
    console.log(count);
    count++;
}

// For...of (duyệt giá trị)
let fruits = ["táo", "chuối", "cam"];
for (let fruit of fruits) {
    console.log(fruit);
}

// For...in (duyệt key)
let person = { name: "An", age: 25 };
for (let key in person) {
    console.log(`${key}: ${person[key]}`);
}

Array Methods quan trọng#

let numbers = [1, 2, 3, 4, 5];

// map - biến đổi mỗi phần tử
let doubled = numbers.map(n => n * 2); // [2, 4, 6, 8, 10]

// filter - lọc phần tử
let evens = numbers.filter(n => n % 2 === 0); // [2, 4]

// reduce - gộp thành một giá trị
let sum = numbers.reduce((acc, n) => acc + n, 0); // 15

// find - tìm phần tử đầu tiên thỏa điều kiện
let found = numbers.find(n => n > 3); // 4

// forEach - duyệt qua mảng
numbers.forEach(n => console.log(n));

Kết luận#

Nắm vững các khái niệm cơ bản này là bước đầu quan trọng để trở thành JavaScript developer giỏi!
