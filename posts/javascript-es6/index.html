ES6+ Features: Những tính năng hiện đại của JavaScript

January 20, 2024 · 4 min · Phú Sơn

ES6 (ECMAScript 2015) đã mang đến nhiều tính năng mạnh mẽ cho JavaScript. Hãy cùng khám phá những tính năng quan trọng nhất.

1. Destructuring - Phá vỡ cấu trúc#

Object Destructuring#

const person = {
    name: "Minh",
    age: 25,
    city: "Hà Nội"
};

// Cách cũ
const name = person.name;
const age = person.age;

// ES6 - ngắn gọn hơn
const { name, age, city } = person;

// Đổi tên biến
const { name: fullName, age: tuoi } = person;

// Giá trị mặc định
const { country = "Việt Nam" } = person;

Array Destructuring#

const colors = ["đỏ", "xanh", "vàng", "tím"];

const [first, second, ...rest] = colors;
console.log(first);  // "đỏ"
console.log(second); // "xanh"
console.log(rest);   // ["vàng", "tím"]

// Skip phần tử
const [, , third] = colors;
console.log(third); // "vàng"

2. Spread Operator (…) - Toán tử rải#

// Copy array
const arr1 = [1, 2, 3];
const arr2 = [...arr1]; // [1, 2, 3]

// Merge arrays
const combined = [...arr1, 4, 5, 6]; // [1, 2, 3, 4, 5, 6]

// Copy object
const obj1 = { a: 1, b: 2 };
const obj2 = { ...obj1, c: 3 }; // { a: 1, b: 2, c: 3 }

// Function arguments
function sum(...numbers) {
    return numbers.reduce((a, b) => a + b, 0);
}
console.log(sum(1, 2, 3, 4, 5)); // 15

3. Template Literals - Chuỗi mẫu#

const name = "An";
const age = 25;

// Cách cũ
const message1 = "Tên tôi là " + name + " và tôi " + age + " tuổi";

// ES6 - dễ đọc hơn
const message2 = `Tên tôi là ${name} và tôi ${age} tuổi`;

// Multi-line
const html = `
    
        ${name}
        Tuổi: ${age}
    
`;

// Expression
const total = `Tổng: ${10 + 20 + 30}`;

4. Enhanced Object Literals#

const name = "Laptop";
const price = 15000000;

// Shorthand property
const product = { name, price }; // { name: "Laptop", price: 15000000 }

// Method shorthand
const calculator = {
    // Cách cũ: add: function(a, b) { return a + b; }
    add(a, b) {
        return a + b;
    },
    subtract(a, b) {
        return a - b;
    }
};

// Computed property names
const key = "dynamicKey";
const obj = {
    [key]: "giá trị",
    [`${key}2`]: "giá trị khác"
};

5. Default Parameters#

// Cách cũ
function greet(name) {
    name = name || "Khách";
    return "Xin chào, " + name;
}

// ES6
function greet(name = "Khách") {
    return `Xin chào, ${name}`;
}

// Với nhiều tham số
function createUser(name, age = 18, role = "user") {
    return { name, age, role };
}

6. Classes - Lập trình hướng đối tượng#

class Person {
    constructor(name, age) {
        this.name = name;
        this.age = age;
    }
    
    // Method
    introduce() {
        return `Tôi là ${this.name}, ${this.age} tuổi`;
    }
    
    // Static method
    static compareAge(person1, person2) {
        return person1.age - person2.age;
    }
}

// Kế thừa
class Student extends Person {
    constructor(name, age, grade) {
        super(name, age);
        this.grade = grade;
    }
    
    study() {
        return `${this.name} đang học lớp ${this.grade}`;
    }
}

const student = new Student("An", 16, 10);
console.log(student.introduce()); // "Tôi là An, 16 tuổi"
console.log(student.study());     // "An đang học lớp 10"

7. Modules - Import/Export#

// math.js
export const PI = 3.14159;

export function add(a, b) {
    return a + b;
}

export default class Calculator {
    multiply(a, b) {
        return a * b;
    }
}

// main.js
import Calculator, { PI, add } from './math.js';
import * as MathUtils from './math.js';

console.log(add(5, 3));
console.log(PI);

8. Optional Chaining (?.) và Nullish Coalescing (??)#

const user = {
    profile: {
        address: {
            city: "Hà Nội"
        }
    }
};

// Cách cũ - dài dòng
const city = user && user.profile && user.profile.address && user.profile.address.city;

// Optional chaining - ngắn gọn
const city2 = user?.profile?.address?.city; // "Hà Nội"
const zip = user?.profile?.address?.zip; // undefined

// Nullish coalescing
const port = null ?? 3000; // 3000
const name = "" ?? "Default"; // "" (vì "" không phải null/undefined)

Kết luận#

ES6+ đã làm JavaScript hiện đại và mạnh mẽ hơn rất nhiều. Hãy sử dụng những tính năng này để viết code clean và hiệu quả!
